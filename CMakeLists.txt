cmake_minimum_required(VERSION 3.20)
project(SE VERSION 0.7.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(s_core
  src/lexer/token.cpp src/lexer/lexer.cpp src/parser/parser.cpp
  src/runtime/error.cpp src/runtime/value.cpp src/runtime/platform_base.cpp src/runtime/advanced.cpp src/runtime/database.cpp src/runtime/expansion.cpp src/modules/modules.cpp src/ffi/ffi.cpp src/ffi/bindgen.cpp
  src/interpreter/interpreter.cpp src/type/checker.cpp src/compiler/compiler.cpp src/compiler/web_compiler.cpp src/compiler/component_web.cpp)
target_include_directories(s_core PUBLIC include)
target_compile_definitions(s_core PRIVATE S_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}" S_NATIVE_BUILD_DIR="${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(s_core PUBLIC ${CMAKE_DL_LIBS})
if(WIN32)
  target_link_libraries(s_core PUBLIC ws2_32)
endif()
if(MSVC)
  target_compile_options(s_core PRIVATE /W4 /WX /permissive- /wd4456 /wd4996)
else()
  target_compile_options(s_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  set_source_files_properties(src/runtime/platform_base.cpp src/runtime/advanced.cpp src/interpreter/interpreter.cpp src/type/checker.cpp PROPERTIES COMPILE_OPTIONS "-Wno-misleading-indentation")
endif()

add_executable(se src/main.cpp)
target_link_libraries(se PRIVATE s_core)
target_compile_definitions(se PRIVATE S_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}" S_NATIVE_BUILD_DIR="${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  target_compile_options(se PRIVATE /W4 /WX /permissive- /wd4456 /wd4996)
else()
  target_compile_options(se PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_library(native_test SHARED native/native_test.c)
target_include_directories(native_test PRIVATE include native)
set_target_properties(native_test PROPERTIES OUTPUT_NAME native_test)
if(MSVC)
  target_compile_options(native_test PRIVATE /W4 /WX /wd4996)
else()
  target_compile_options(native_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
